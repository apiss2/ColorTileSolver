cmake_minimum_required(VERSION 3.16)
project(ColorTileSolver VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_library(colortile_core STATIC src/solver.cpp src/recognition.cpp)
target_include_directories(colortile_core PUBLIC src)
target_link_libraries(colortile_core PUBLIC Threads::Threads)
add_executable(ColorTileSolverCLI src/cli.cpp)
target_link_libraries(ColorTileSolverCLI PRIVATE colortile_core)
if(WIN32)
    if(NOT MINGW)
        message(FATAL_ERROR "Windows builds require a 64-bit LLVM-MinGW or MinGW-w64 toolchain.")
    endif()
    enable_language(RC)
    add_executable(ColorTileSolver WIN32 src/winmain.cpp src/resources.rc)
    target_include_directories(ColorTileSolver PRIVATE src)
    # サンプルは実行ファイルの隣から読み込むため、ビルド先にも配置する。
    add_custom_command(TARGET ColorTileSolver POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:ColorTileSolver>/examples"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "${CMAKE_CURRENT_SOURCE_DIR}/examples/sample.png"
            "$<TARGET_FILE_DIR:ColorTileSolver>/examples/sample.png"
        VERBATIM)
    target_link_libraries(ColorTileSolver PRIVATE colortile_core windowscodecs ole32 comctl32 comdlg32 shell32 gdi32 uuid)
    target_link_options(ColorTileSolver PRIVATE -municode -static -Wl,--stack,8388608)
    target_link_options(ColorTileSolverCLI PRIVATE -municode -static -Wl,--stack,8388608)
    add_executable(test_gui_windows tests/test_gui_windows.cpp)
    target_link_options(test_gui_windows PRIVATE -municode -static)
    target_link_libraries(test_gui_windows PRIVATE user32)
endif()
enable_testing()
add_executable(test_core tests/test_core.cpp)
target_link_libraries(test_core PRIVATE colortile_core)
add_test(NAME core_reference COMMAND test_core "${CMAKE_CURRENT_SOURCE_DIR}/examples/sample.ctb")
add_executable(test_solver_exhaustive tests/test_solver_exhaustive.cpp)
target_link_libraries(test_solver_exhaustive PRIVATE colortile_core)
add_test(NAME solver_exhaustive COMMAND test_solver_exhaustive)
add_executable(test_recognition tests/test_recognition.cpp)
target_link_libraries(test_recognition PRIVATE colortile_core)
add_test(NAME recognition_validation COMMAND test_recognition --self-test)
